cmake_minimum_required(VERSION 3.24)
project(odin_slicer LANGUAGES C CXX)

# Use via: cmake -DCMAKE_TOOLCHAIN_FILE=cmake/iOS.cmake -S . -B build-ios
#                -DCMAKE_PROJECT_FILE=CMakeLists.ios.txt ...
# (CMake 3.24+ required for nested CMAKE_PROJECT_FILE.)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# ---------------------------------------------------------------------------
# Stage 1: stub build — produces `odin_slicer_stub.a` that implements the
# C API with `NOT_AVAILABLE` returns. This is what ships in ODIN Studio today.
# Stage 2 (separate track): drop in the real libslic3r dep graph and replace
# the source in this target.
# ---------------------------------------------------------------------------

add_library(odin_slicer_stub STATIC
    src/odin_slicer_api.cpp
)

target_include_directories(odin_slicer_stub PUBLIC
    ${CMAKE_SOURCE_DIR}/include
)

# Explicit flag to satisfy the iOS toolchain's stripped sdk — no Foundation
# dependency at this layer.
set_target_properties(odin_slicer_stub PROPERTIES
    OUTPUT_NAME odin_slicer_stub
    XCODE_ATTRIBUTE_ENABLE_BITCODE NO
)

message(STATUS "odin_slicer: building stub (libslic3r integration = pending)")
